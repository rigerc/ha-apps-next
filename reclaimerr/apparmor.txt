#include <tunables/global>

profile reclaimerr flags=(attach_disconnected,mediate_deleted) {
  # The image is a Python/FastAPI service.  These abstractions cover only the
  # common read-only runtime files and resolver/TLS support it needs.
  #include <abstractions/base>
  #include <abstractions/nameservice>
  #include <abstractions/openssl>
  #include <abstractions/ssl_certs>

  # Root may need to access a host media directory whose numeric owner differs
  # from the container.  No UID-transition, SYS_ADMIN, or raw-device
  # capability is granted by this profile.
  capability dac_override,

  network inet stream,
  network inet dgram,
  network inet6 stream,
  network inet6 dgram,
  network unix stream,
  network unix dgram,
  signal (send, receive) set=(term,int,hup,kill,cont),

  /bin/bash rix,
  /bin/busybox rix,
  /bin/sh rix,
  /bin/dash rix,
  /bin/mkdir rix,
  /docker-entrypoint.sh rix,
  /run.sh rix,
  /sbin/su-exec rix,
  /usr/bin/env rix,
  /usr/bin/bash rix,
  /usr/bin/curl rix,
  /usr/bin/gosu rix,
  /usr/bin/mkdir rix,
  /usr/bin/python3* rix,
  /usr/bin/su-exec rix,
  /usr/local/bin/docker-entrypoint.sh rix,
  /usr/local/bin/granian rix,
  /usr/local/bin/gosu rix,
  /usr/local/bin/python* rix,

  # The packaged backend and built SPA are immutable at runtime.  The m bit
  # is needed for Python bytecode/module mappings and is not write access.
  /app/ rm,
  /app/** rm,

  /dev/null rw,
  /dev/random r,
  /dev/urandom r,
  /dev/shm/ rw,
  /dev/shm/** rwk,
  /etc/** r,
  /lib/** rm,
  /proc/** r,
  /sys/devices/system/cpu/online r,
  /tmp/ rwk,
  /tmp/** rwk,
  /usr/lib/** rm,
  /usr/local/lib/** rm,
  /usr/share/** r,

  # HA supplies options.json; no other /data subtree is visible.
  /data/ rw,
  /data/options.json r,
  /data/reclaimerr/ rwkl,
  /data/reclaimerr/** rwkl,

  # The app's only host data mapping.  Keep the mount root writable so the
  # wrapper can create its explicitly scoped fixture/path-mapping children.
  /media/ rw,
  /media/** rwkl,

  /run/ rwk,
  /run/** rwk,
}
